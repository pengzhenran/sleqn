# PyInstaller 版本资源：写进 exe 属性页的"详细信息"
#
# 在 spec 里用 version='packaging/version_info.txt' 引用。
# 语言用 0x0804（简体中文）+ 代码页 1200（Unicode），所以 StringTable 的键是
# '080404B0'，VarFileInfo 的 Translation 也必须是 (0x0804, 1200)，两者要一致，
# 否则资源管理器里读不到。

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 1, 0),
    prodvers=(1, 0, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '080404B0',
        [StringStruct('CompanyName', '中国地质大学（武汉）'),
         StringStruct('FileDescription', '海平面指纹（SLF）计算程序'),
         StringStruct('FileVersion', '1.0.1.0'),
         StringStruct('InternalName', 'sleqn'),
         StringStruct('LegalCopyright',
                      'Copyright (C) 2026 彭桢燃 (Zhenran Peng). '
                      '本程序依据 LGPL-3.0 使用 Qt for Python (PySide6)。'),
         StringStruct('OriginalFilename', 'sleqn.exe'),
         StringStruct('ProductName', '海平面指纹（SLF）计算程序'),
         StringStruct('ProductVersion', '1.0.1.0'),
         StringStruct('Comments',
                      '重力数据反演的海平面指纹计算工具；'
                      '算法：Sun et al. (2022), Pure Appl. Geophys. 179(9).')],
      )
    ]),
    VarFileInfo([VarStruct('Translation', [0x0804, 1200])]),
  ],
)
